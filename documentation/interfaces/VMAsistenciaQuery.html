export type AsistenciaPeriodoTipo =
  | 'week'
  | 'month'
  | 'year'
  | 'range';

export type AsistenciaView = 'day' | 'month';

export type AsistenciaTablaSegmento =
  | 'anteriores'
  | 'hoy'
  | 'proximos'
  | 'all';
