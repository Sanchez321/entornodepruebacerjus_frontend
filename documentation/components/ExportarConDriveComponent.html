import { CommonModule } from '@angular/common';
import { Component, EventEmitter, Input, Output } from '@angular/core';

@Component({
  selector: 'app-exportar-con-drive',
  standalone: true,
  imports: [CommonModule],
  templateUrl: './exportar-con-drive.html',
  styleUrl: './exportar-con-drive.css',
})
export class ExportarConDriveComponent {
  @Input() label = 'Descargar';
  @Input() loadingLabel = 'Preparando…';
  @Input() helperText = '';
  @Input() disabled = false;
  @Input() loading = false;

  @Output() exportar = new EventEmitter<boolean>();

  guardarDrive = false;

  emitir(): void {
    if (this.disabled || this.loading) return;
    this.exportar.emit(this.guardarDrive);
  }
}

<div class="export-drive-wrap">
  <label class="export-drive-check">
    <input
      #saveCheck
      type="checkbox"
      [checked]="guardarDrive"
      [disabled]="disabled || loading"
      (change)="guardarDrive = saveCheck.checked">
    <span>Guardar también en Drive</span>
  </label>

  <button
    type="button"
    class="btn btn-primary"
    [disabled]="disabled || loading"
    (click)="emitir()">
    {{ loading ? loadingLabel : label }}
  </button>

  @if (helperText) {
    <span class="text-muted small">{{ helperText }}</span>
  }
</div>
